ESP32 & 5 Ultrasons

DISPOSITION

Capteur1: Disposé en avant sur un cerveau moteur et pouvant donc périodiquement effectuée une demi rotation jouant un peu le rôle de Radar

Capteur2: En avant sous le châssis et tourné vers le sol

Capteur3: Au milieu du coté gauche et orthogonal au plan du châssis

Capteur4: Au milieu du coté droit et orthogonal au plan du châssis

Capteur5: En arrière

BRANCHEMENT

Capteur 1: TRIG → GPIO 15 ECHO → GPIO 2

Capteur 2: TRIG → GPIO 4 ECHO → GPIO 16

Capteur 3: TRIG → GPIO 17 ECHO → GPIO 5

Capteur 4 : TRIG → GPIO 19 ECHO → GPIO 18

Capteur 5: TRIG → GPIO 21 ECHO → GPIO 34

                  VCC→5 V

                  GND→GND

CODE POUR LIRE LES 5 CAPTEURS 


// ---------------------------
// Définition des broches TRIG
// ---------------------------
#define TRIG1 15   // Broche TRIG du capteur 1
#define TRIG2 4   // Broche TRIG du capteur 2
#define TRIG3 17    // Broche TRIG du capteur 3
#define TRIG4 19   // Broche TRIG du capteur 4
#define TRIG5 21  // Broche TRIG du capteur 5

// ---------------------------
// Définition des broches ECHO
// ---------------------------

#define ECHO1 2   // Broche ECHO du capteur 1
#define ECHO2 16   // Broche ECHO du capteur 2
#define ECHO3 5    // Broche ECHO du capteur 3
#define ECHO4 18   // Broche ECHO du capteur 4
#define ECHO5 34   // Broche ECHO du capteur 5

// --------------------------------------------------
// Fonction qui mesure la distance avec un capteur
// --------------------------------------------------

long lireDistance(int trigPin, int echoPin)
{

digitalWrite(trigPin, LOW);      // Met la broche TRIG à 0
delayMicroseconds(2);            // Petite pause pour stabiliser le capteur

digitalWrite(trigPin, HIGH);     // Envoie une impulsion ultrason
delayMicroseconds(10);           // Durée de l'impulsion (10 µs)
digitalWrite(trigPin, LOW);      // Arrêt de l'impulsion

long duree = pulseIn(echoPin, HIGH);   // Mesure le temps du signal reçu
long distance = duree * 0.034 / 2;     // Conversion du temps en distance (cm)

return distance;                 // Renvoie la distance mesurée
}

// ---------------------------
// Configuration initiale
// ---------------------------

void setup()
{

Serial.begin(115200); // Démarre la communication série avec le PC

// Configuration des broches TRIG en sortie

pinMode(TRIG1, OUTPUT); // TRIG1 est une sortie
pinMode(TRIG2, OUTPUT); // TRIG2 est une sortie
pinMode(TRIG3, OUTPUT); // TRIG3 est une sortie
pinMode(TRIG4, OUTPUT); // TRIG4 est une sortie
pinMode(TRIG5, OUTPUT); // TRIG5 est une sortie

// Configuration des broches ECHO en entrée

pinMode(ECHO1, INPUT);  // ECHO1 est une entrée
pinMode(ECHO2, INPUT);  // ECHO2 est une entrée
pinMode(ECHO3, INPUT);  // ECHO3 est une entrée
pinMode(ECHO4, INPUT);  // ECHO4 est une entrée
pinMode(ECHO5, INPUT);  // ECHO5 est une entrée


}



// ---------------------------
// Programme principal
// ---------------------------



void loop()
{



long d1 = lireDistance(TRIG1, ECHO1); // Mesure la distance du capteur 1
delay(60);                            // Pause pour éviter interférences



long d2 = lireDistance(TRIG2, ECHO2); // Mesure la distance du capteur 2
delay(60);                            // Pause entre capteurs



long d3 = lireDistance(TRIG3, ECHO3); // Mesure la distance du capteur 3
delay(60);



long d4 = lireDistance(TRIG4, ECHO4); // Mesure la distance du capteur 4
delay(60);



long d5 = lireDistance(TRIG5, ECHO5); // Mesure la distance du capteur 5
delay(60);



// Affichage des distances dans le moniteur série



Serial.print("Capteur 1 : ");
Serial.print(d1);
Serial.println(" cm");




Serial.print("Capteur 2 : ");
Serial.print(d2);
Serial.println(" cm");


Serial.print("Capteur 3 : ");
Serial.print(d3);
Serial.println(" cm");


Serial.print("Capteur 4 : ");
Serial.print(d4);
Serial.println(" cm");


Serial.print("Capteur 5 : ");
Serial.print(d5);
Serial.println(" cm");


Serial.println("---------------------------");



delay(500); // Attente avant la prochaine lecture
}
