Capteur4:(Right) Au milieu du coté droit et orthogonal au plan du châssis

🧠 🔵 LOGIQUE DE NAVIGATION (PSEUDO-LIGNE INTELLIGENTE)

Le robot adopte une stratégie de balayage progressif de la pièce, inspirée du zigzag, mais adaptée à l’absence de capteurs de position précis. Il avance globalement en ligne droite tout en corrigeant sa trajectoire en temps réel grâce aux capteurs latéraux. Lorsqu’un mur ou un obstacle est détecté à gauche ou à droite, il ajuste légèrement la vitesse de ses roues pour rester à distance constante, ce qui permet de maintenir une trajectoire quasi rectiligne (pseudo-ligne).
En parallèle, le capteur ultrason monté sur servo effectue un balayage continu (radar) pour anticiper les obstacles frontaux. Lorsqu’un obstacle est détecté devant, le robot s’arrête, analyse rapidement l’espace à gauche et à droite à l’aide du servo, puis effectue un demi-tour ou un changement de direction en se décalant latéralement avant de repartir dans le sens opposé. Ce décalage progressif permet de reproduire un comportement de type zigzag, assurant ainsi une couverture de surface efficace.
Les capteurs orientés vers le sol (avant et arrière) assurent une sécurité permanente : si un vide est détecté (escalier), le robot interrompt immédiatement son mouvement et effectue une manœuvre d’évitement adaptée (recul ou avance + rotation). Le capteur arrière empêche également toute collision lors des phases de recul.
Ainsi, le robot combine :
une progression en pseudo-ligne stabilisée par correction latérale,
un changement de bande automatique lors des obstacles,
un balayage radar pour anticiper,
et une gestion prioritaire de la sécurité,
ce qui permet d’obtenir une couverture progressive, robuste et adaptée à un environnement réel sans nécessiter de capteurs avancés de position.

🔵CODE LOGIQUE

#include <Arduino.h>
#include <ESP32Servo.h>

// ----------------------
// CAPTEURS
// ----------------------
#define TRIG_FRONT 5
#define ECHO_FRONT 18

#define TRIG_DOWN_FRONT 19
#define ECHO_DOWN_FRONT 21

#define TRIG_DOWN_BACK 22
#define ECHO_DOWN_BACK 23

#define TRIG_BACK 25
#define ECHO_BACK 26

#define TRIG_LEFT 27
#define ECHO_LEFT 14

#define TRIG_RIGHT 32
#define ECHO_RIGHT 33

// ----------------------
// MOTEURS L298N
// ----------------------
#define ENA 12
#define ENB 13
#define IN1 16
#define IN2 4
#define IN3 2
#define IN4 17

// ----------------------
#define SERVO_PIN 15
Servo radar;

// ----------------------
// VARIABLES TEMPS
unsigned long tServo = 0;
unsigned long tSensor = 0;
unsigned long tDecision = 0;
unsigned long tShift = 0;

// ----------------------
int angle = 0;
int dir = 1;

// distances
float front, downF, downB, back, leftD, rightD;

// mode zigzag
bool changementLigne = false;

// ----------------------
// DISTANCE
float lireDist(int trig, int echo) {
  digitalWrite(trig, LOW);
  delayMicroseconds(2);
  digitalWrite(trig, HIGH);
  delayMicroseconds(10);
  digitalWrite(trig, LOW);

  long d = pulseIn(echo, HIGH, 30000);
  if (d == 0) return 999;

  return d * 0.034 / 2;
}

// ----------------------
// MOTEUR PWM
void moteur(int g, int d) {

  ledcWrite(0, abs(g));
  ledcWrite(1, abs(d));

  digitalWrite(IN1, g > 0);
  digitalWrite(IN2, g <= 0);

  digitalWrite(IN3, d > 0);
  digitalWrite(IN4, d <= 0);
}

// ----------------------
void setup() {
  Serial.begin(115200);

  // capteurs
  pinMode(TRIG_FRONT, OUTPUT); pinMode(ECHO_FRONT, INPUT);
  pinMode(TRIG_DOWN_FRONT, OUTPUT); pinMode(ECHO_DOWN_FRONT, INPUT);
  pinMode(TRIG_DOWN_BACK, OUTPUT); pinMode(ECHO_DOWN_BACK, INPUT);
  pinMode(TRIG_BACK, OUTPUT); pinMode(ECHO_BACK, INPUT);
  pinMode(TRIG_LEFT, OUTPUT); pinMode(ECHO_LEFT, INPUT);
  pinMode(TRIG_RIGHT, OUTPUT); pinMode(ECHO_RIGHT, INPUT);

  // moteurs
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);


  ledcAttach(ENA,1000,8);
  ledcAttach(ENB,1000,8);

  radar.attach(SERVO_PIN);
}

// ----------------------
void loop() {

  unsigned long now = millis();

  // 🔄 RADAR CONTINU
  if (now - tServo > 40) {
    tServo = now;

    radar.write(angle);
    angle += dir * 5;

    if (angle >= 180 || angle <= 0)
      dir *= -1;
  }

  // 📡 LECTURE CAPTEURS
  if (now - tSensor > 80) {
    tSensor = now;

    front = lireDist(TRIG_FRONT, ECHO_FRONT);
    downF = lireDist(TRIG_DOWN_FRONT, ECHO_DOWN_FRONT);
    downB = lireDist(TRIG_DOWN_BACK, ECHO_DOWN_BACK);
    back  = lireDist(TRIG_BACK, ECHO_BACK);
    leftD = lireDist(TRIG_LEFT, ECHO_LEFT);
    rightD= lireDist(TRIG_RIGHT, ECHO_RIGHT);
  }

  // 🧠 DÉCISION
  if (now - tDecision > 100) {
    tDecision = now;

    // 🔴 SÉCURITÉ VIDE AVANT
    if (downF > 50) {
      moteur(-150, -150);
      return;
    }

    // 🔴 SÉCURITÉ VIDE ARRIÈRE
    if (downB > 50) {
      moteur(150, 150);
      return;
    }

    // 🔴 OBSTACLE DEVANT
    if (front < 25) {
      changementLigne = true;
      tShift = now;
    }

    // 🟣 CHANGEMENT DE "LIGNE" (zigzag)
    if (changementLigne) {

      // petit recul si possible
      if (back > 20 && now - tShift < 300) {
        moteur(-150, -150);
        return;
      }

      // décalage latéral
      if (now - tShift < 800) {
        moteur(150, -150); // rotation
        return;
      }

      changementLigne = false;
    }

    // 🟢 PSEUDO-LIGNE (correction latérale)
    int base = 170;

    if (leftD < 15) {
      moteur(base + 30, base - 40); // corrige droite
    }
    else if (rightD < 15) {
      moteur(base - 40, base + 30); // corrige gauche
    }
    else {
      moteur(base, base); // ligne droite
    }

    // 🔵 EXPLORATION (zone ouverte)
    if (leftD > 50 && rightD > 50) {
      moteur(base, base - 20); // légère courbe
    }
  }
}
